~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ vss-a24-l24 building router vss ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

! ########### VRF START ########### 
!
ip vrf {VRF_name}
  rd 64000:{RD_ID}
  route-target export 64000:{RD_ID}
  route-target import 64000:{RD_ID}
!
router bgp 64000
  address-family ipv4 vrf {VRF_name}
    redistribute connected
    redistribute static
    no synchronization
        network 0.0.0.0
        exit-address-family
!
! ########### VRF END #############
