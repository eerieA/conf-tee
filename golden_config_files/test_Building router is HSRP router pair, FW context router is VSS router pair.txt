~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ site01-hub1 building router hsrp ~~~~~~~~~~~~~~~~~~~~~

! ########### VRF START ########### 
!
ip vrf {VRF_name}
  rd 64000:{RD_ID}
  route-target export 64000:{RD_ID}
  route-target import 64000:{RD_ID}
!
router bgp 64000
  address-family ipv4 vrf {VRF_name}
    redistribute connected
    redistribute static
    no synchronization
        network 0.0.0.0
        exit-address-family
!
! ########### VRF END ############# 


! ########### USER PREFIX START ########### 
!
vlan {Vlan_ID_user}
  name {Vlan_name}
!
interface GigabitEthernet2/28
 switchport trunk allowed vlan add {Vlan_ID_user}
!
interface vlan{Vlan_ID_user}
  description {Vlan_name}
  mtu 9216
  ip vrf forwarding {VRF_name}
  ip address {{User_Prefix_last_IP}-1} {User_prefix_netmask}
  ip helper-address 142.103.1.1
  ip helper-address 10.32.57.4
  standby ip  {User_Prefix_last_IP}
  standby preempt
  standby priority 105
  standby authentication md5 key-string 0 ewhr324jflewfjosdfjofds
  no shutdown
!
! ########### USER PREFIX END #############
