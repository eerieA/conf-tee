~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ site01-hub1 FW context and building router hsrp ~~~~~~~~~~~~~~~~~~~~~~~~~~

! ########### USER PREFIX START ########### 
!
vlan {Vlan_ID_user}
  name {Vlan_name}
!
interface {int_name_A}{int_#}
 switchport trunk allowed vlan add {Vlan_ID_user}
!
interface vlan{Vlan_ID_user}
  description {Vlan_name}
  mtu 9216
  ip vrf forwarding {VRF_name}
  ip address {{User_Prefix_last_IP}-1} {User_prefix_netmask}
  ip helper-address {IP_helper_address}
  standby ip  {User_Prefix_last_IP}
  standby preempt
  standby priority 105
  standby authentication md5 key-string 0 {key_word}
  no shutdown
!
ip route {User_prefix} {User_prefix_netmask} {GW-FW_outside_IP} name {FW_context_name}:{Vlan_name}
!
!copy running startup
!
! ########### USER PREFIX END ############# 

! ########### MULTICAST START ########### 
!
ip multicast-routing vrf {VRF_name}
!
ip vrf {VRF_name}
  mdt default {mdt_default_address}
  mdt data {mdt_data_address} 255.255.255.192 threshold 10
  mdt log-reuse
!
interface Loopback{RD_#}0000
  description RP Loopback for VRF {VRF_name}
  ip vrf forwarding {VRF_name}
  ip address 10.239.0.0 255.255.255.255
  ip pim sparse-mode
!
ip pim vrf {VRF_name} rp-address 10.239.0.0
!
interface Loopback{RD_#}0001
  description MSDP Loopback for VRF {VRF_name}
  ip vrf forwarding {VRF_name}
  ip address 10.239.0.1 255.255.255.255
  ip pim sparse-mode
!
ip msdp vrf {VRF_name} peer 10.239.0.2 connect-source Loopback{RD_#}0001
ip msdp vrf {VRF_name} cache-sa-state
ip msdp vrf {VRF_name} originator-id Loopback{RD_#}0001
!
interface Vlan{Vlan_ID_user}
  ip pim sparse-mode
!
!copy running start
!
