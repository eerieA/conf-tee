~~~~~~~~~~~~~~~~~~~~~~~~~ site01-hub1 VPN gateway router hsrp ~~~~~~~~~~~~~~~~~~~~~

! ########### VRF START ########### 
!
ip vrf {VRF_name}
  rd 64000:{RD_ID}
  route-target export 64000:{RD_ID}
  route-target import 64000:{RD_ID}
!
router bgp 64000
  address-family ipv4 vrf {VRF_name}
    redistribute connected
    redistribute static
    no synchronization
        network 0.0.0.0
        exit-address-family
!
! ########### VRF END ############# 

! ########### VGLUE PREFIX START ########### 
!
vlan {Vlan_ID_vpnglue}
  name VPNX01-{IP_local_pool_name}-VGLUE={VRF_name}
!
interface {int_name_A}{int_#}
 switchport trunk allowed vlan add {Vlan_ID_vpnglue}
!
interface Port-channel1
 switchport trunk allowed vlan add {Vlan_ID_vpnglue}
!
interface vlan{Vlan_ID_vpnglue}
  description VPNX01-{IP_local_pool_name}-VGLUE={VRF_name}
  mtu 9216
  ip vrf forwarding {VRF_name}
  ip address {{Vpnglue_prefix_last_IP}-1} 255.255.255.248
  standby ip {Vpnglue_prefix_last_IP}
  standby preempt
  standby priority 105
  standby authentication md5 key-string 0 {key_word}
  no shutdown
!
ip route vrf {VRF_name} {IP_local_pool_prefix} {IP_local_pool_netmask} {Vpnglue_prefix_1st_IP} name {FW_context_name}:VPNX01-{IP_local_pool_name}={VRF_name}
!
! ########### VPN PREFIX END #############

~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ myvpn vpn gateway ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

! ########### VPN PREFIX START ########### 
config terminal
!
interface Ethernet1/8.{Vlan_ID_vpnglue}
  nameif {IP_local_pool_name}
  security-level 50
  ip address {Vpnglue_prefix_1st_IP} 255.255.255.248 standby {{Vpnglue_prefix_1st_IP}+1}
!
mtu {IP_local_pool_name} 1500
!
ip local pool {IP_local_pool_name} {IP_local_pool_first_IP}-{IP_local_pool_last_IP} mask {IP_local_pool_netmask}
route {IP_local_pool_name} 0.0.0.0 0.0.0.0 {Vpnglue_prefix_last_IP} {IBVPN_metric}
!
group_policy {Vlan_ID_vpnglue} internal
group_policy {Vlan_ID_vpnglue} attributes
  split-tunnel-policy tunnelall
  vlan {Vlan_ID_vpnglue}
  address-pools value {IP_local_pool_name}
exit
!
